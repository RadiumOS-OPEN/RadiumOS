# Bochs configuration for RadiumOS
# Modified to boot from os.iso (CD-ROM) with higher VRAM

# Memory: 256 MB
megs: 256

# CPU: 1 core
cpu: count=1, ips=10000000

# BIOS and VGA BIOS
romimage: file=/usr/share/bochs/BIOS-bochs-latest
vgaromimage: file=/usr/share/bochs/VGABIOS-lgpl-latest

# Boot from CD-ROM
boot: c
ata0: enabled=1, ioaddr1=0x1f0, ioaddr2=0x3f0, irq=14
ata0-master: type=cdrom, path=os.iso, status=inserted

# VGA: Standard with VBE extension and 16MB memory
vga: vbe=on update_freq=5 vgamem=16777216

# Display: SDL
display_library: sdl

# Serial: To terminal
com1: enabled=1, mode=term

# Log file
log: bochs.log

# Debug options
debug: action=report

# Mouse
mouse: enabled=1

# Keyboard
keyboard: type=mf, serial_delay=250, paste_delay=100000

# Clock sync
clock: sync=realtime
